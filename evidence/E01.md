def procesar_factura(event, context):
    """
    Se invoca cuando se sube un archivo (p.ej. .pdf o .jpg) al bucket INPUT_BUCKET.
    Solo se ejecutará si PROCESS_MODE == "onupload".
    """
    if PROCESS_MODE != "onupload":
        print("[INFO] Trigger 'onupload' desactivado. Usar modo cron/manual.")
        return

    file_name = event["name"]  # p.ej. "user_001/doc_abc.jpg"
    gcs_input_uri = f"gs://{INPUT_BUCKET}/{file_name}"

    # Comprobar MIME type
    blob = storage.Client().bucket(INPUT_BUCKET).blob(file_name)
    mime_type = blob.content_type or guess_mime_type(file_name)

    if not mime_type:
        raise ValueError(f"[ERROR] Tipo MIME no válido para el archivo: {file_name}")

    print(f"[PROCESS] (onupload) Procesando archivo: {file_name} (MIME: {mime_type})")

    # Llamada principal a Document AI (solo 1 archivo en este modo)
    batch_process_documents_single(
        gcs_input_uri=gcs_input_uri,
        input_mime_type=mime_type
    )
