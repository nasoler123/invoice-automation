def rest_submit(container: str, rel_path: str) -> str:
    api  = (
        f"{ENDPOINT}/documentintelligence/documentModels/{MODEL_ID}:"
        f"analyze?api-version={API_VER}"
    )
    body = json.dumps({"urlSource": url_blob(container, rel_path, SAS_PROCESSED)}).encode()
    hdrs = {
        "Ocp-Apim-Subscription-Key": AZ_KEY,
        "Content-Type":              "application/json",
        "Accept":                    "application/json",
    }
    for i in range(RETRIES + 1):
        try:
            r = requests.post(api, headers=hdrs, data=body, timeout=30)
            if r.status_code == 202:
                return r.headers["Operation-Location"]
            if r.status_code in (429, 500, 503):
                raise RuntimeError(f"{r.status_code} {r.text[:120]}")
            r.raise_for_status()
        except Exception as e:
            if i == RETRIES:
                raise
            sleep = (2 ** i) + random.random()
            logging.warning(
                "[RETRY %d/%d] %s → %s (sleep %.1fs)",
                i + 1, RETRIES, rel_path, e, sleep
            )
            time.sleep(sleep)
    raise RuntimeError("retries exhausted")
