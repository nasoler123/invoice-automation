def check_line_level(ln:Dict[str,Any]) -> (bool,str):
    """
    Verifica:
      - base_value, tax_value, rate_value, total_line no sean nulos
      - total_line ~ base_val + tax_val (dentro de un margen)
      - tax_val ~ base_val*(rate_val/100) (margen)
    """
    base_val = ln.get("base_value", None)
    tax_val  = ln.get("tax_value", None)
    rate_val = ln.get("rate_value", None)
    tot_line = ln.get("total_line", None)
    
    # 1) Campos vacíos
    if base_val is None or tax_val is None or rate_val is None or tot_line is None:
        return True, "DatosLineaIncompletos"
    
    # Convertir a float
    base_val = float(base_val)
    tax_val  = float(tax_val)
    rate_val = float(rate_val)
    tot_line = float(tot_line)
    
    # 2) total_line ~ base_val + tax_val
    if not approx_equal(tot_line, base_val + tax_val, tol=0.001):
        return True, "SumaLineasNoCuadra"
    
    # 3) base_val * (rate_val/100) ~ tax_val
    expected_tax = base_val * (rate_val/100.0)
    if not approx_equal(tax_val, expected_tax, tol=0.01):
        return True, "IVANoCuadraConRate"
    
    return False, ""


def approx_equal(a: float, b: float, tol: float=0.001)->bool:
    return abs(a - b) <= tol
