def create_invoice_in_holded(header, lines, holded_key, cuenta_map:dict):
    """
    - Lee validacion_fr => (validacion_holded, motivo, holded_cuenta_contable, tax_tag_holded)
    - Aplica "Revisar" si hay discrepancias y al menos una línea con 'Revisar'
    - Crea **un ítem por cada línea** (no unifica).
    - 'accountingAccountId' => se asigna con cuenta_map[hcc] (ID real en Holded)
    - 'taxes' => [ tax_tag_holded ] si lo hay

    invoice_str = header.get("invoice_date","") or ""
    invoice_ts = date_to_timestamp(invoice_str)
    if not invoice_ts:
        invoice_ts = int(dt.now().timestamp())

    user_id = header["user_id"]
    invoice_id = header["invoice_id"]

    bq_client = get_bigquery_client()
    sql_val = f"""
    SELECT id, validacion_holded, motivo, holded_cuenta_contable, tax_tag_holded
    FROM {PROJECT_ID}.{DATASET_ID}.{VALIDACION_FR_TABLE}
    WHERE user_id=@u_id
      AND invoice_id=@inv
    """
    job_c = bigquery.QueryJobConfig(
        query_parameters=[
            bigquery.ScalarQueryParameter("u_id","STRING",user_id),
            bigquery.ScalarQueryParameter("inv","STRING",invoice_id),
        ]
    )
    val_rows = list(bq_client.query(sql_val, job_c).result())
    val_map = {}
    for vr in val_rows:
        val_map[vr.id] = {
            "validacion_holded": vr.validacion_holded,
            "motivo": vr.motivo,
            "hcc": vr.holded_cuenta_contable,
            "tag": vr.tax_tag_holded
        }
