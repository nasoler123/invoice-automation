def compute_hist_embeddings(hist: dict[str, list[str]]) -> dict[str, list[list[float]]]:
    """Return embeddings for each provider description."""
    return {cid: [MODEL.encode(t) for t in descs] for cid, descs in hist.items()}


def embed_sim(desc: str, cid: str, hist_embed: dict[str, list[list[float]]]) -> float:
    if cid not in hist_embed:
        return 0.0
    emb = MODEL.encode(desc)
    return float(max(util.cos_sim(emb, hist_embed[cid]).tolist()[0]))
